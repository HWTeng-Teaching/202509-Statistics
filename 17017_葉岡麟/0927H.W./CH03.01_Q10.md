•  男女比例很接近：男生 52% 有手機、48% 沒有；女生 55% 有手機、45% 沒有。
•  女生擁有手機的比例略高（55% vs. 52%），相對地沒手機的比例略低（45% vs. 48%）。
•  整體差異不大；從描述統計看，兩者分布相似，只是女生稍微更常擁有手機。 
import numpy as np
import matplotlib.pyplot as plt

# 資料
groups = ['Boys', 'Girls']
cell = np.array([52, 55])     # 有手機
no_cell = np.array([48, 45])  # 沒手機

# ---------- a) 堆疊長條圖 ----------
x = np.arange(len(groups))
fig, ax = plt.subplots(figsize=(8, 5), dpi=150)

bars_cell = ax.bar(x, cell, label='Cell Phone')
bars_nocell = ax.bar(x, no_cell, bottom=cell, label='No Cell Phone')

ax.set_xticks(x, groups)
ax.set_ylabel('Number of students')
ax.set_title('Cell Phone Ownership (8th grade): Stacked')
ax.legend()

# 在柱狀圖上標數字 python
for b in bars_cell:
    h = b.get_height()
    ax.text(b.get_x() + b.get_width()/2, h/2, f'{int(h)}',
            ha='center', va='center', fontsize=9)
for b, base in zip(bars_nocell, cell):
    h = b.get_height()
    ax.text(b.get_x() + b.get_width()/2, base + h/2, f'{int(h)}',
            ha='center', va='center', fontsize=9)

plt.tight_layout()
plt.savefig('stacked_bar.png')
plt.show()

# ---------- b) 並列長條圖 ----------
fig, ax = plt.subplots(figsize=(8, 5), dpi=150)
width = 0.35

bars_cell = ax.bar(x - width/2, cell, width, label='Cell Phone')
bars_nocell = ax.bar(x + width/2, no_cell, width, label='No Cell Phone')

ax.set_xticks(x, groups)
ax.set_ylabel('Number of students')
ax.set_title('Cell Phone Ownership (8th grade): Side-by-Side')
ax.legend()

# 在柱狀圖上標數字
for b in list(bars_cell) + list(bars_nocell):
    h = b.get_height()
    ax.text(b.get_x() + b.get_width()/2, h + 0.5, f'{int(h)}',
            ha='center', va='bottom', fontsize=9)

ax.set_ylim(0, max(cell.max(), no_cell.max()) + 10)
plt.tight_layout()
plt.savefig('side_by_side_bar.png')
plt.show()

# ---------- (附加) c) 基本比較輸出 ----------
boys_total = cell[0] + no_cell[0]
girls_total = cell[1] + no_cell[1]
boys_pct_cell = cell[0] / boys_total * 100
girls_pct_cell = cell[1] / girls_total * 100

print(f"Boys with cell phone:  {boys_pct_cell:.1f}%")
print(f"Girls with cell phone: {girls_pct_cell:.1f}%")
print(f"Difference (Girls - Boys): {girls_pct_cell - boys_pct_cell:.1f} percentage points")
