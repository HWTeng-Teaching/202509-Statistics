import matplotlib.pyplot as plt

# 資料
labels = ['男生', '女生']
have_phone = [52, 55]
no_phone = [48, 45]

# 畫堆疊長條圖
fig, ax = plt.subplots(figsize=(6, 5))
ax.bar(labels, have_phone, label='有手機', color='#4f81bd')
ax.bar(labels, no_phone, bottom=have_phone, label='沒有手機', color='#c0504d')

ax.set_ylabel('學生人數')
ax.set_xlabel('性別')
ax.set_title('八年級學生手機擁有情況（堆疊長條圖）')
ax.legend()

plt.tight_layout()
plt.show()
