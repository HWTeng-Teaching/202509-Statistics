### The code:<br>
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np

# Original data
original_data = pd.DataFrame({
    'Temperature': [21, 21, 23, 24, 28, 28, 29, 28, 33, 31],
    'Chirps_per_second': [14, 15, 16, 16, 16, 17, 18, 18, 19, 20]
})

# Regression coefficients
b = 0.41
a = 6.12

# Create the plot
plt.figure(figsize=(10, 6))

# Generate smooth regression line
temperature_range = np.linspace(20, 34, 100)
chirps_predicted = a + b * temperature_range

# Plot original data points
plt.scatter(original_data['Temperature'], original_data['Chirps_per_second'], 
           s=80, color='blue', alpha=0.7, edgecolor='black', linewidth=1,
           label='Observed Data')

# Plot regression line
plt.plot(temperature_range, chirps_predicted, 'r-', linewidth=2,
         label=f'Regression Line: y = {a:.2f} + {b:.2f}x')

# Customize the plot
plt.xlabel('Temperature', fontsize=12)
plt.ylabel('Chirps per Second', fontsize=12)
plt.title('Temperature vs Cricket Chirps Frequency', fontsize=14)
plt.grid(True, alpha=0.3, linestyle='--')
plt.legend(fontsize=10)

# Set axis limits and ticks
plt.xlim(20, 34)
plt.ylim(13, 21)
plt.xticks(range(20, 35, 2))
plt.yticks(range(13, 22, 1))

plt.tight_layout()
plt.show()

# Print summary
print("REGRESSION SUMMARY")
print("=" * 40)
print(f"Equation: y = {a:.2f} + {b:.2f}x")
print(f"Slope: {b:.2f} chirps/sec per °C")
print(f"Intercept: {a:.2f} chirps/sec")<br>
